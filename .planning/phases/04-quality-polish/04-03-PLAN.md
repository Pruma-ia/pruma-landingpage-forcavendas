    ```bash
    cat > /tmp/overflow-check.js << 'SCRIPT'
    const { chromium } = require('playwright');

    (async () => {
      const browser = await chromium.launch();
      const viewports = [
        { width: 375, height: 812, name: 'mobile-375' },
        { width: 768, height: 1024, name: 'tablet-768' },
        { width: 1024, height: 768, name: 'desktop-1024' },
        { width: 1440, height: 900, name: 'desktop-1440' },
      ];

      for (const vp of viewports) {
        const page = await browser.newPage();
        await page.setViewportSize({ width: vp.width, height: vp.height });
        await page.goto('http://localhost:3000', { waitUntil: 'networkidle' });

        const docWidth = await page.evaluate(() => document.documentElement.scrollWidth);
        const overflow = docWidth > vp.width;
        console.log(`${vp.name}: docWidth=${docWidth}px viewport=${vp.width}px overflow=${overflow}`);

        if (overflow) {
          const culprits = await page.evaluate((vpWidth) => {
            const els = document.querySelectorAll('*');
            const offenders = [];
            for (const el of els) {
              const rect = el.getBoundingClientRect();
              if (rect.right > vpWidth + 2) {
                offenders.push({
                  tag: el.tagName,
                  id: el.id || '',
                  classes: Array.from(el.classList).slice(0, 4).join(' '),
                  right: Math.round(rect.right),
                });
              }
            }
            return offenders.slice(0, 10);
          }, vp.width);
          console.log(`  Offenders:`, JSON.stringify(culprits, null, 2));
        }

        // Screenshot
        await page.screenshot({ path: `/tmp/screenshot-${vp.name}.png`, fullPage: false });
        await page.close();
      }
      await browser.close();
    })();
    SCRIPT
    node /tmp/overflow-check.js
    ```

    CORREÇÃO OBRIGATÓRIA — Hero.tsx: Mockup PainelDashboard em mobile
    O Hero já tem `overflow-x-auto -mx-6 px-6` na div do mockup. Verifique se funciona.

    ```tsx
        <PainelDashboard />
      </div>
    </motion.div>
    ```

    CORREÇÃO OBRIGATÓRIA — Funcionalidades.tsx: Bento hero card com scale(0.72)
    O container do PainelDashboard escalado tem `height: 320px` mas o elemento interno
    pode transbordar horizontalmente antes da escala. Envolva com `overflow-hidden`:
    ```tsx
    // Linha ~62, a div de overflow deve ser:
    <div className="overflow-hidden w-full" style={{ height: "320px" }}>
      <div style={{ transform: "scale(0.72)", transformOrigin: "top center" }}>
        <PainelDashboard />
      </div>
    </div>
    ```

    CORREÇÃO OBRIGATÓRIA — globals.css: Prevenir overflow global
    Adicione ao bloco `html` em globals.css:
    ```css
    html {
      color-scheme: light;
      overflow-x: hidden; /* Previne scroll horizontal de conteúdo fora do fluxo */
    }
    ```
    ATENÇÃO: `overflow-x: hidden` no `html` é último recurso e esconde sintomas sem corrigir
    causas. Use-o como guard de segurança APÓS corrigir as causas específicas identificadas acima.

    CORREÇÃO CONDICIONAL — outros componentes:
    Para qualquer componente que o audit reportar como `overflow=true`:
    - Se tem grid: adicionar `min-w-0` nos grid children
    - Se tem flex: verificar `flex-wrap` e `min-w-0`
    - Se tem mockup/SVG: adicionar `overflow-hidden` no container pai
    - Se tem texto longo: adicionar `overflow-wrap: break-word` via `break-words` class
